    root = Path(os.environ.get("CEDAR_ROOT", "<배포 후 생성기 위치>"))       # 이동 전: /Users/sdw79/SDWjavis/_레포/keunsaem-preview/r4/drafts/cedar
    repo = root  # 커밋 대상 저장소 루트로 바꿔 줄 것(이동 후 ksmc31)
    for args in ([], ["--sub"]):
        r = subprocess.run(["python3", str(root / "_build.py"), *args], capture_output=True, text=True)
        if r.returncode != 0: return notify_master("[막힘] cedar 재생성 실패\n" + r.stderr[-800:])

    if r.returncode != 0 or "parity=False" in r.stdout or "complete=False" in r.stdout or "broken=['" in r.stdout or "'overflow': True" in r.stdout:

    g = lambda *c: subprocess.run(["git", "-C", str(repo), *c], capture_output=True, text=True)

def notify_master(msg):
